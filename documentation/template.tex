% documentation/template.tex — pandoc-template for the ZeroDDS doc-trail.

%       --template=documentation/template.tex \
%       --pdf-engine=tectonic

% The template is intentionally minimal — branded heading + hyperref +
% one accent colour. Station-specific LaTeX (e.g. architecture.tex) bypasses
% this template and is built directly.

\usepackage[margin=2.4cm]{geometry}
\usepackage{fontspec}
\usepackage{microtype}
\usepackage{xcolor}
\usepackage{listings}

  UprightFont = *,
  BoldFont    = *-Bold,
